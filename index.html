                extrasArray.push(`Topping: ${topExtra} (+$10)`);

    function eliminarItem(index) {

        carrito.splice(index, 1);

        renderCarrito();

    }


    function renderCarrito() {

        const tbody = document.getElementById("carritoBody");

        tbody.innerHTML = "";

        let total = 0;


        carrito.forEach((item, index) => {

            total += item.subtotal;

            tbody.innerHTML += `

                <tr>

                    <td><strong>${item.producto}</strong></td>

                    <td>${item.sabor_base}</td>

                    <td>${item.tamano}</td>

                    <td>${item.fruta}</td>

                    <td>${item.topping_extra}</td>

                    <td class="text-center">${item.cantidad}</td>

                    <td class="text-right">$${item.precio.toFixed(2)}</td>

                    <td class="text-right">$${item.subtotal.toFixed(2)}</td>

                    <td class="text-center">

                        <button class="btn-delete" onclick="eliminarItem(${index})">❌</button>

                    </td>

                </tr>

            `;

        });


        document.getElementById("totalDisplay").textContent = `TOTAL: $${total.toFixed(2)}`;

    }


    async function registrarEnGoogleSheets() {

        if (carrito.length === 0) {

            alert("El carrito está vacío.");

            return;

        }


        // Calcular insumos requeridos en esta venta

        let reqVortices = 0;

        let reqCookies = 0;


        carrito.forEach(item => {

            if (item.producto === "Vórtice") {

                reqVortices += item.cantidad;

            } else if (item.producto === "Cookies Sandwich") {

                reqCookies += item.cantidad;

            }

        });


        // Validar si hay suficiente stock

        if (reqVortices > stockVortices) {

            alert(`⚠️ Stock insuficiente de Masa para Vórtices. Disponibles: ${stockVortices}, Requeridos: ${reqVortices}`);

            return;

        }


        if (reqCookies > stockCookies) {

            alert(`⚠️ Stock insuficiente de Masa para Cookies. Disponibles: ${stockCookies}, Requeridos: ${reqCookies}`);

            return;

        }


        if (URL_GOOGLE_SHEETS.includes("AQUI_PEGA_TU_URL")) {

            alert("Recuerda pegar tu URL de Google Apps Script.");

            return;

        }


        const btn = document.getElementById("btnCobrar");

        btn.disabled = true;

        btn.textContent = "⏳ Guardando...";


        const metodoPago = document.getElementById("metodoPagoSelect").value;

        const fechaHora = new Date().toLocaleString("es-MX");


        const payload = {

            items: carrito.map(item => ({

                id_venta: `V-${idVenta}`,

                fecha: fechaHora,

                producto: item.producto,

                sabor_base: item.sabor_base,

                tamano: item.tamano,

                fruta: item.fruta,

                topping_extra: item.topping_extra,

                cantidad: item.cantidad,

                precio: item.precio,

                subtotal: item.subtotal,

                metodo_pago: metodoPago

            }))

        };


        try {

            await fetch(URL_GOOGLE_SHEETS, {

                method: "POST",

                mode: "no-cors",

                headers: { "Content-Type": "application/json" },

                body: JSON.stringify(payload)

            });


            // Descontar del inventario

            stockVortices -= reqVortices;

            stockCookies -= reqCookies;


            localStorage.setItem("pos_stock_vortices", stockVortices.toString());

            localStorage.setItem("pos_stock_cookies", stockCookies.toString());

            renderStock();


            alert(`✅ Venta V-${idVenta} guardada en Google Sheets. Se actualizaron los inventarios.`);

            idVenta++;

            localStorage.setItem("pos_id_venta_gs", idVenta.toString());

            carrito = [];

            renderCarrito();

        } catch (error) {

            alert("❌ Ocurrió un error al enviar la venta: " + error);

        } finally {

            btn.disabled = false;

            btn.textContent = "☁️ Cobrar y Guardar en Google Sheets";

        }

    }
